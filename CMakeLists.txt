cmake_minimum_required(VERSION 3.22)

# mdit — a fast, pure-Qt6 specialty markdown editor with a live preview.
# Pure Qt only: no QtWebEngine, no QtMultimedia, no external markdown library.
project(mdit
    VERSION 0.1.0
    DESCRIPTION "A fast, pure-Qt6 specialty markdown editor with a live preview."
    LANGUAGES CXX)

# ---------------------------------------------------------------------------
# C++ standard: C++20 (no GNU extensions).
# ---------------------------------------------------------------------------
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# ---------------------------------------------------------------------------
# Qt6. PrintSupport is required for PDF export (QPdfWriter); Widgets/Gui for the
# GUI + the markdown renderer (QTextDocument::setMarkdown). AUTOMOC/AUTORCC/
# AUTOUIC are enabled by the qt_* helpers, but set them explicitly so plain
# add_executable() targets in tests/ also pick up moc-processed headers.
# ---------------------------------------------------------------------------
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)

find_package(Qt6 COMPONENTS Widgets Gui PrintSupport REQUIRED)

# Sensible default build type for single-config generators (Ninja/Make).
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
    set_property(CACHE CMAKE_BUILD_TYPE PROPERTY STRINGS
        Debug Release RelWithDebInfo MinSizeRel)
endif()

# ---------------------------------------------------------------------------
# Testing.
# ---------------------------------------------------------------------------
enable_testing()
option(BUILD_TESTS "Build the mdit CTest/QtTest suite." ON)

# ---------------------------------------------------------------------------
# Main executable.
#
# Sources live in src/ and are globbed so newly added module files (MainWindow,
# Document, EditorPane, PreviewPane, RenderedDocument, Theme, Settings,
# MarkdownModel, MarkdownHighlighter) are picked up automatically as later
# subtasks add them. CONFIGURE_DEPENDS re-runs the glob when files change.
# ---------------------------------------------------------------------------
# All app sources live in src/. main.cpp is the entry point only; the rest are
# the library of modules (MainWindow, Document, EditorPane, PreviewPane,
# RenderedDocument, Theme, Settings, MarkdownModel, MarkdownHighlighter) added by
# the subtasks. Splitting them into a static lib lets the CTest binaries link the
# same module code without re-implementing it.
file(GLOB MDIT_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
list(FILTER MDIT_SOURCES EXCLUDE REGEX ".*/main\\.cpp$")

# The application icon (assets/icons/*.png, generated by tools/make_icon.py) is
# compiled into mdit_core through this .qrc (AUTORCC), so `:/icons/mdit-*.png`
# is available at runtime without shipping loose files.
list(APPEND MDIT_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/resources/mdit.qrc")

add_library(mdit_core STATIC ${MDIT_SOURCES})
target_include_directories(mdit_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
# AppInfo::version() reads this (PUBLIC so the tests see the same version string
# as the app — one source of truth: project(mdit VERSION ...) above).
target_compile_definitions(mdit_core PUBLIC MDIT_VERSION="${PROJECT_VERSION}")
target_link_libraries(mdit_core PUBLIC
    Qt6::Widgets
    Qt6::Gui
    Qt6::PrintSupport
)

qt_add_executable(mdit WIN32 MACOSX_BUNDLE
    "${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp")
target_link_libraries(mdit PRIVATE mdit_core)

# Windows .exe (no console) / macOS .app bundle metadata.
set_target_properties(mdit PROPERTIES
    WIN32_EXECUTABLE TRUE
    MACOSX_BUNDLE TRUE
    MACOSX_BUNDLE_BUNDLE_NAME "mdit"
    MACOSX_BUNDLE_GUI_IDENTIFIER "io.wendorf.mdit"
    MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
)

# ---------------------------------------------------------------------------
# Tests (C++ QtTest driven by CTest).
# ---------------------------------------------------------------------------
if(BUILD_TESTS)
    add_subdirectory(tests)
endif()

# ---------------------------------------------------------------------------
# Install (optional; a Linux launcher integration)
#
# `cmake --install build` puts the binary on PATH, the icon into the hicolor
# theme (so a launcher/taskbar shows it) and a .desktop entry next to it.
# ---------------------------------------------------------------------------
include(GNUInstallDirs)
install(TARGETS mdit
        RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}"
        # macOS builds a .app bundle (MACOSX_BUNDLE): CMake refuses to configure
        # an install(TARGETS) for a bundle without a BUNDLE DESTINATION, even
        # though our .dmg comes from build_macos.sh rather than `cmake --install`.
        BUNDLE DESTINATION "${CMAKE_INSTALL_BINDIR}")
foreach(size IN ITEMS 32 48 64 128 256 512)
    install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/assets/icons/mdit-${size}.png"
            DESTINATION "${CMAKE_INSTALL_DATADIR}/icons/hicolor/${size}x${size}/apps"
            RENAME mdit.png)
endforeach()
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/packaging/mdit.desktop"
        DESTINATION "${CMAKE_INSTALL_DATADIR}/applications")
